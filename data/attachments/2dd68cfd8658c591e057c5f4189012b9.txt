var response = pm.response.json(); 

// Validar que el status sea 200
pm.test("Status code es 200", function () {
    pm.response.to.have.status(200);
});

// Validar que la respuesta sea en formato JSON
pm.test("Respuesta es JSON", function () {
    pm.response.to.be.json;
});

pm.test("Validar esquema de respuesta", function () {
    const schema = {
        type: "object",
        required: [
            "Id",
            "Content",
            "ItemType",
            "Checked",
            "ProjectId",
            "ParentId",
            "Path",
            "Collapsed",
            "DateString",
            "DateStringPriority",
            "DueDate",
            "Recurrence",
            "ItemOrder",
            "Priority",
            "LastSyncedDateTime",
            "Children",
            "DueDateTime",
            "CreatedDate",
            "LastCheckedDate",
            "LastUpdatedDate",
            "Deleted",
            "Notes",
            "InHistory",
            "SyncClientCreationId",
            "DueTimeSpecified",
            "OwnerId"
        ],
        properties: {
            Id: { type: "number" },
            Content: { type: "string" },
            ItemType: { type: "number" },
            Checked: { type: "boolean" },
            ProjectId: { type: "number" },
            ParentId: { type: ["number", "null"] },
            Path: { type: "string" },
            Collapsed: { type: "boolean" },
            DateString: { type: ["string", "null"] },
            DateStringPriority: { type: "number" },
            DueDate: { type: "string" },
            Recurrence: { type: ["object", "null"] },
            ItemOrder: { type: "number" },
            Priority: { type: "number" },
            LastSyncedDateTime: { type: "string" },
            Children: { type: "array" },
            DueDateTime: { type: ["string", "null"] },
            CreatedDate: { type: "string" },
            LastCheckedDate: { type: ["string", "null"] },
            LastUpdatedDate: { type: "string" },
            Deleted: { type: "boolean" },
            Notes: { type: "string" },
            InHistory: { type: "boolean" },
            SyncClientCreationId: { type: ["string", "null"] },
            DueTimeSpecified: { type: "boolean" },
            OwnerId: { type: "number" }
        }
    };

    pm.response.to.have.jsonSchema(schema);
});

// Validar nombre de item actualizado
pm.test("Verificar nombre de item actualizado", function () {
    var Content = response.Content
    pm.expect(Content).to.eql(pm.collectionVariables.get("newItemName"));
});

// Validar checked de item actualizado
pm.test("Verificar checked actualizado", function () {
    const checkedItemR = response.Checked;
    const checkedItemC = JSON.parse(
        pm.collectionVariables.get("checkedItem")
    );

    pm.expect(checkedItemR).to.eql(checkedItemC);
});
// Validar item en el projecto correcto
pm.test("Verificar item en el projecto correcto", function () {
    var projectIdr = response.ProjectId; // project id del response
    let projectIdc = parseInt(pm.collectionVariables.get("projectId"), 10); // project id de la coleccion
    pm.expect(projectIdr).to.eql(projectIdc);
});

// Validar que item existente
pm.test("Verficiar item, no debe estar eliminado", () => {
    pm.expect(response.Deleted).to.equal(false);
});

// Validar fechas con formatos correctos
pm.test("Verificar fechas con formato Microsoft", () => {
    const regex = /^\/Date\(\d+\)\/$/;

    pm.expect(response.CreatedDate).to.match(regex);
    pm.expect(response.LastUpdatedDate).to.match(regex);
    pm.expect(response.LastSyncedDateTime).to.match(regex);
});

pm.test("Verificar existencia y formato de LastCheckedDate cuando el item cambia a estado Done", () => {
    const regex = /^\/Date\(\d+\)\/$/;
    if (response.Checked) {
        pm.expect(response.LastCheckedDate).to.not.be.null;
        pm.expect(response.CreatedDate).to.match(regex);
    }
});
