var response = pm.response.json(); 

// Validar que la respuesta sea en formato JSON
pm.test("Respuesta es JSON", function () {
    pm.response.to.be.json;
});
 
// Validar que el status sea 200
pm.test("Status code es 200", function () {
    pm.response.to.have.status(200);
    var id = response.Id; 
    pm.collectionVariables.set("projectId", id);
});

pm.test("Schema de error válido", () => {
    const schema = {
        type: "object",
        required: ["ErrorMessage", "ErrorCode"],
        properties: {
            ErrorMessage: {
                type: "string"
            },
            ErrorCode: {
                type: "number"
            }
        }
    };

    pm.response.to.have.jsonSchema(schema);
});

pm.test("ErrorCode es 102", () => {
    pm.expect(response.ErrorCode).to.eql(102);
});

pm.test("Mensaje: Not Authenticated", () => {
    pm.expect(response.ErrorMessage).to.eql("Not Authenticated");
});
